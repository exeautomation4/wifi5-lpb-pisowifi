#!/usr/bin/env python3
"""
WiFi5 LPB Piso WiFi - GPIO Coin Detection & 12V Cut-Off Relay Daemon
Compatible with: Orange Pi (OPi.GPIO / gpiod) and Raspberry Pi (RPi.GPIO)
Listens for pulse trains from Allan 1234A / 1235A Multi-Coin Acceptors.
"""

import time
import sys
import os
import threading
import json
import urllib.request

# Configuration
PULSE_PIN = 12       # Orange Pi PA12 (Physical Pin 3)
RELAY_PIN = 11       # 12V Coin Acceptor Cut-Off Relay (Physical Pin 5)
DEBOUNCE_MS = 40     # Allan 1234A standard pulse width
INTER_COIN_GAP = 0.5 # Seconds of silence before finalizing pulse count

pulse_counter = 0
last_pulse_time = 0
lock = threading.Lock()

print("[WIFI5] Initializing GPIO Coin Detection Daemon...")

def on_pulse_detected():
    """Triggered by hardware interrupt on falling edge."""
    global pulse_counter, last_pulse_time
    now = time.time()
    with lock:
        # Software Debounce Filter
        if (now - last_pulse_time) * 1000 >= DEBOUNCE_MS:
            pulse_counter += 1
            last_pulse_time = now
            print(f"[PULSE] Detected pulse #{pulse_counter}")

def pulse_evaluator_loop():
    """Timer loop that counts pulses and converts to Philippine Pesos."""
    global pulse_counter, last_pulse_time
    while True:
        time.sleep(0.1)
        with lock:
            if pulse_counter > 0 and (time.time() - last_pulse_time) > INTER_COIN_GAP:
                amount_peso = pulse_counter
                print(f"[COIN CREDITED] Finalized: {pulse_counter} pulses = ₱{amount_peso}.00")
                notify_local_server(amount_peso)
                pulse_counter = 0

def notify_local_server(peso):
    """Notifies local backend server of coin insertion."""
    url = f"http://127.0.0.1:8080/api/coin-inserted?amount={peso}"
    try:
        req = urllib.request.Request(url, method='POST')
        with urllib.request.urlopen(req, timeout=2) as response:
            print(f"[API] Server acknowledged ₱{peso} coin credit.")
    except Exception as e:
        print(f"[API ERROR] Failed to notify local server: {e}")

if __name__ == "__main__":
    t = threading.Thread(target=pulse_evaluator_loop, daemon=True)
    t.start()
    print("[WIFI5] Coin Listener Active. Ready for coins.")
    while True:
        time.sleep(1)
